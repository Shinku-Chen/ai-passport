cmake_minimum_required(VERSION 3.16)
project(voice_marquee_test C)
set(LVGL_DIR "${CMAKE_CURRENT_LIST_DIR}/../../managed_components/lvgl__lvgl" CACHE PATH "Resolved LVGL 9.5.0 source")
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
if(NOT LVGL_SOURCES)
    message(FATAL_ERROR "Resolve the project's managed LVGL dependency before building this test")
endif()
add_executable(test_voice_marquee test_voice_marquee.c
    ../../main/voice_marquee.c ../../main/fonts/voice_cjk.c ${LVGL_SOURCES})
target_include_directories(test_voice_marquee PRIVATE . ../../main "${LVGL_DIR}")
target_compile_definitions(test_voice_marquee PRIVATE LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE=1)
if(NOT MSVC)
    target_link_libraries(test_voice_marquee PRIVATE m)
endif()
enable_testing()
add_test(NAME voice_marquee COMMAND test_voice_marquee)
