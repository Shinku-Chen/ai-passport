# GalGame reader build: the framework UI/pagination/settings plus the
# title-specific packed reader. The placeholder gal_app/content tables are not
# compiled.
idf_component_register(
    SRCS "main.c"
         "gal/gal_text.c"
         "gal/gal_settings.c"
         "gal/gal_ui.c"
         "nine/nine_assets.c"
         "nine/nine_crc.c"
         "nine/nine_pack.c"
         "nine/nine_vm.c"
         "nine/nine_image.c"
         "nine/nine_media.c"
         "nine/nine_op_log.c"
         "nine/nine_store.c"
         "nine/nine_reader.c"
         "nine/nine_app.c"
         "nine/tjpgd/tjpgd.c"
    INCLUDE_DIRS "." "gal" "nine" "nine/tjpgd"
    REQUIRES bsp lvgl esp_timer nvs_flash esp_hw_support esp_partition
)

# The reader is landscape-only; gal_ui keeps its portrait defaults otherwise.
target_compile_definitions(${COMPONENT_LIB} PRIVATE NINE_READER=1)

# CJK subsets for all story/UI glyphs (generated by tools/nine_font.py).
target_sources(${COMPONENT_LIB} PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/../assets/nine/nine_cjk_16.c"
    "${CMAKE_CURRENT_LIST_DIR}/../assets/nine/nine_cjk_20.c"
)
