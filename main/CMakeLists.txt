# 9-nine- 九次九日九重色 reader. The framework UI/text modules stay shared; the
# title provides its own pack reader, VM, compositor, store and application.
set(NINE_PACK_DIR "${CMAKE_CURRENT_LIST_DIR}/../assets/nine/kokoiro")
set(NINE_SCRIPT_BIN "${NINE_PACK_DIR}/nine_script.bin")
set(NINE_IMAGES_BIN "${NINE_PACK_DIR}/nine_images.bin")
if(NOT EXISTS "${NINE_SCRIPT_BIN}" OR NOT EXISTS "${NINE_IMAGES_BIN}")
    message(FATAL_ERROR "Missing 9-nine- pack. Run: python tools/nine_pack.py "
                        "--source <9-nine-rw5 checkout> --episode kokoiro "
                        "--out assets/nine/kokoiro")
endif()
set(NINE_ASSETS_ASM "${CMAKE_CURRENT_BINARY_DIR}/nine_assets.S")
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    add_custom_command(OUTPUT "${NINE_ASSETS_ASM}"
        COMMAND ${CMAKE_COMMAND}
                "-DTEMPLATE=${CMAKE_CURRENT_LIST_DIR}/nine/nine_assets.S.in"
                "-DOUT=${NINE_ASSETS_ASM}"
                "-DNINE_SCRIPT_BIN=${NINE_SCRIPT_BIN}"
                "-DNINE_IMAGES_BIN=${NINE_IMAGES_BIN}"
                -P "${CMAKE_CURRENT_LIST_DIR}/nine/gen_assets.cmake"
        DEPENDS "${NINE_SCRIPT_BIN}" "${NINE_IMAGES_BIN}" "${CMAKE_CURRENT_LIST_DIR}/nine/nine_assets.S.in"
        VERBATIM)
endif()
idf_component_register(
    SRCS "main.c"
         "gal/gal_text.c"
         "gal/gal_settings.c"
         "gal/gal_tone.c"
         "gal/gal_music.c"
         "gal/gal_ui.c"
         "nine/nine_app.c"
         "nine/nine_pack.c"
         "nine/nine_vm.c"
         "nine/nine_store.c"
         "nine/nine_image.c"
         "nine/nine_media.c"
         "nine/tjpgd/tjpgd.c"
         "${NINE_ASSETS_ASM}"
    INCLUDE_DIRS "." "gal" "nine"
    REQUIRES bsp lvgl esp_timer nvs_flash esp_hw_support
)
target_compile_definitions(${COMPONENT_LIB} PRIVATE NINE_READER=1)

# 9-nine- 16/20px subset fonts (generated by tools/nine_font.py).
target_sources(${COMPONENT_LIB} PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/../assets/nine/kokoiro/nine_cjk_16.c"
    "${CMAKE_CURRENT_LIST_DIR}/../assets/nine/kokoiro/nine_cjk_20.c"
)
